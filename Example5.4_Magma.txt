clear;
Z := IntegerRing();

function RelTrace(a, m, F)
    t := F!0;
    for i in [0..m-1] do t +:= a^(2^i); end for;
    return t;
end function;

procedure HullCase(label, m, poly, e1, e2, e3)
    n := 2*m;
    F := ext<GF(2) | poly>;
    xi := F.1;
    lambda1 := F!1; lambda2 := xi^e1;
    w1 := xi^e2; w2 := xi^e3;
    N := 2^n-1;
    S := [x : x in F | x ne 0];
    B := Basis(F);
    G := ZeroMatrix(GF(2), n+1, N);
    for i in [1..n] do
        for j in [1..N] do G[i,j] := Z!Trace(B[i]*S[j]); end for;
    end for;
    for j in [1..N] do
        x := S[j]; xp := x^(2^m+1);
        tm1 := Z!RelTrace(lambda1*xp, m, F);
        tm2 := Z!RelTrace(lambda2*xp, m, F);
        g := (tm1*tm2) mod 2;
        ind := 0;
        if x eq w1 then ind := 1; end if;
        if x eq w2 then ind := 1; end if;
        G[n+1,j] := (g+ind) mod 2;
    end for;
    C := LinearCode(G);
    Hull := C meet Dual(C);
    printf "%o\n", label;
    printf "  C_f      : [%o, %o, %o]\n", Length(C), Dimension(C), MinimumWeight(C);
    printf "  Hull(C_f): [%o, %o, %o]\n",
           Length(Hull), Dimension(Hull), MinimumWeight(Hull);
    printf "\n";
end procedure;

P<x> := PolynomialRing(GF(2));
HullCase("Example 5.4 (Example 5.1): m=3", 3,
         x^6 + x^4 + x^3 + x + 1, 9, 1, 22);
HullCase("Example 5.4 (Example 5.2): m=6", 6,
         x^12 + x^7 + x^6 + x^5 + x^3 + x + 1, 1170, 2034, 3204);
HullCase("Example 5.4 (Example 5.3): m=5", 5,
         x^10 + x^6 + x^5 + x^3 + x^2 + x + 1, 33, 15, 356);
