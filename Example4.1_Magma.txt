//     (1) n=3, d=3, xi^3+xi+1=0;
//     (2) n=4, d=3, xi^4+xi+1=0;

// FIELD
//   All cases: F_{2^n} generated by the primitive element xi with the

//   prescribed for that case.
//
// VERIFIED OUTPUT (reproduces Example 4.1)
//   (1) W_f = -4 (1 time), 0 (4), 4 (3); F(xi^5)=1; wt(f)=4; C_f [7,4,2];

//   (2) W_f = -8 (1), 0 (12), 8 (3); F = 0; wt(f)=12; C_f [15,5,4];
//       Hull(C_f) [15,5,4];
//   (3) W_f = -4 (4), 0 (6), 4 (4), 8 (2); F(xi^3)=1; wt(f)=8; C_f [15,5,4];
//       Hull(C_f) [15,3,8].

//   Response 13 (supplementary Magma package), item "reproduces the
//   numerical data of the examples in Sections 4-6".

clear;
Z := IntegerRing();

procedure RunCase(label, n, d, F, e_test)
    N := 2^n-1;
    S := [x : x in F | x ne 0];
    B := Basis(F);
    G := ZeroMatrix(GF(2), n+1, N);
    for i in [1..n] do
        for j in [1..N] do G[i,j] := Z!Trace(B[i]*S[j]); end for;
    end for;
    for j in [1..N] do G[n+1,j] := Z!Trace(S[j]^d); end for;
    C := LinearCode(G);
    Hull := C meet Dual(C);

    // Walsh spectrum of f(x) = Tr_1^n(x^d)
    vals := [];
    for beta in F do
        s := 0;
        for x in F do
            s +:= ((Z!Trace(x^d) + Z!Trace(beta*x)) mod 2 eq 0) select 1 else -1;
        end for;
        Append(~vals, s);
    end for;
    V := Sort(vals);
    P := []; i := 1;
    while i le #V do
        j := i;
        while j lt #V and V[j+1] eq V[i] do j +:= 1; end while;
        Append(~P, <V[i], j-i+1>);
        i := j+1;
    end while;

    // wt(f) = #{x in F : f(x) = 1}
    wtf := 0;
    for x in F do
        if Z!Trace(x^d) eq 1 then wtf +:= 1; end if;
    end for;

    // F(b) = sum_{x in F*} f(x) Tr_1^n(bx) (mod 2)
    Fzero := true;
    for b in F do
        s := 0;
        for x in F do s +:= (Z!Trace(x^d))*(Z!Trace(b*x)); end for;
        if s mod 2 ne 0 then Fzero := false; break; end if;
    end for;
    Ftest := 0;
    if e_test ge 0 then
        b := F.1^e_test;
        s := 0;
        for x in F do s +:= (Z!Trace(x^d))*(Z!Trace(b*x)); end for;
        Ftest := s mod 2;
    end if;

    printf "%o\n", label;
    printf "  wt(f) = %o\n", wtf;
    printf "  Walsh spectrum:\n";
    for p in P do printf "    W_f = %o : %o time(s)\n", p[1], p[2]; end for;
    if e_test ge 0 then
        printf "  F(xi^%o) = %o\n", e_test, Ftest;
    end if;
    printf "  F = 0 on F_{2^n}: %o\n", Fzero;
    printf "  C_f : [%o, %o, %o]\n", Length(C), Dimension(C), MinimumWeight(C);
    printf "  Hull(C_f) : [%o, %o, %o]\n",
           Length(Hull), Dimension(Hull), MinimumWeight(Hull);
    printf "\n";
end procedure;

P<x> := PolynomialRing(GF(2));
F3 := ext<GF(2) | x^3 + x + 1>;
RunCase("Example 4.1(1): n=3, d=3", 3, 3, F3, 5);

F4d := GF(2^4);
RunCase("Example 4.1(2): n=4, d=3", 4, 3, F4d, -1);

F4e := ext<GF(2) | x^4 + x + 1>;
RunCase("Example 4.1(3): n=4, d=7", 4, 7, F4e, 3);
