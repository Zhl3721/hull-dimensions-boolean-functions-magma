// FIELD (as stated in the example)
//   F_{2^10} with primitive element xi satisfying
//   xi^10 + xi^6 + xi^5 + xi^3 + xi^2 + xi + 1 = 0.
//
// OBJECT
//   f(x) = Tr_1^m(lambda1 x^{2^m+1}) Tr_1^m(lambda2 x^{2^m+1})
//          + (x+w1)^{2^n-1} + (x+w2)^{2^n-1},  m=5, n=10,
//   for the 13 quadruples (lambda1,lambda2,w1,w2) =
//   (1, xi^[t1], xi^[t2], xi^[t3]) listed in the array exps below.
//   The tuple (B1,B2,B3,B4) of each quadruple is reported in Example 5.3.
//
// VERIFIED OUTPUT (reproduces Example 5.3)
//   For every quadruple: C_f = [1023, 11, 266], hull dimension 9,
//   and the weight enumerator listed in Example 5.3.
//
// RESPONSE LETTER
//   Response 13 (supplementary Magma package), item "reproduces all the
//   examples in Sections 4-6".

clear;
m := 5; n := 2*m;
P<x> := PolynomialRing(GF(2));
F<xi> := ext<GF(2) | x^10 + x^6 + x^5 + x^3 + x^2 + x + 1>;

function Trace_n(x)
    t := GF(2)!0;
    for i in [0..n-1] do t +:= x^(2^i); end for;
    return t;
end function;
function Trace_m(x)
    t := GF(2)!0;
    for i in [0..m-1] do t +:= x^(2^i); end for;
    return t;
end function;

exps := [ [33,15,356],[363,16,357],[33,7,348],
          [99,30,371],[33,14,355],[99,25,366],
          [33,1,342],[99,1,342],[33,28,369],
          [363,19,360],[33,27,368],[33,29,370],[99,29,370] ];
S := [x : x in F | x ne 0];
B := Basis(F);
for t in exps do
    lambda1 := F!1; lambda2 := xi^(t[1]);
    w1 := xi^(t[2]); w2 := xi^(t[3]);
    G := ZeroMatrix(GF(2), n+1, #S);
    for i in [1..n] do
        for j in [1..#S] do G[i,j] := Trace_n(B[i]*S[j]); end for;
    end for;
    for j in [1..#S] do
        x := S[j]; xp := x^(2^m+1);
        g := (IntegerRing()!Trace_m(lambda1*xp) * IntegerRing()!Trace_m(lambda2*xp)) mod 2;
        ind := 0;
        if x eq w1 then ind := 1; end if;
        if x eq w2 then ind := 1; end if;
        G[n+1,j] := (g + ind) mod 2;
    end for;
    C := LinearCode(G);
    printf "tuple %o: [%o, %o, %o], hull dim %o\n",
           t, Length(C), Dimension(C), MinimumWeight(C), Dimension(C meet Dual(C));
    wd := WeightDistribution(C);
    for item in Sort([<t0[1], t0[2]> : t0 in wd | t0[2] ne 0]) do
        printf "    weight %o : %o\n", item[1], item[2];
    end for;
end for;
